cmake_minimum_required(VERSION 3.20)
project(karry VERSION 0.2.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
include(GNUInstallDirs)
find_package(Qt6 REQUIRED COMPONENTS Widgets)
foreach(m CoreAddons I18n WidgetsAddons)
    find_package(KF6${m} REQUIRED)
endforeach()
add_executable(karry src/main.cpp src/cli.cpp src/window.cpp src/welcome.cpp src/preview.cpp)
target_compile_definitions(karry PRIVATE KARRY_VERSION="${PROJECT_VERSION}")
target_link_libraries(karry PRIVATE Qt6::Widgets KF6::CoreAddons KF6::I18n KF6::WidgetsAddons)
install(TARGETS karry RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES org.kde.karry.desktop DESTINATION ${CMAKE_INSTALL_DATADIR}/applications)
install(FILES org.kde.karry.svg DESTINATION ${CMAKE_INSTALL_DATADIR}/icons/hicolor/scalable/apps)
